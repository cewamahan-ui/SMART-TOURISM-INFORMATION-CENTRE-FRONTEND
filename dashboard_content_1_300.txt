   214	function CreateAttractionModal({ open, onOpenChange, form, setForm, destinations, submitting, onSubmit }) {
   215	  // Cascading selectors state
   216	  const [county, setCounty] = useState("");
   217	  const [subCounty, setSubCounty] = useState("");
   218	  const [ward, setWard] = useState("");
   219	  const [gps, setGps] = useState("");
       
   220	  // Update cascading options
   221	  const countyObj = COUNTY_DATA.find((c) => c.name === county);
   222	  const subCountyObj = countyObj?.subCounties.find((s) => s.name === subCounty);
   223	  const wardObj = subCountyObj?.wards.find((w) => w.name === ward);
       
   224	  // When ward changes, auto-fill GPS
   225	  React.useEffect(() => {
   226	    if (wardObj) {
   227	      setGps(wardObj.gps);
   228	    }
   229	  }, [ward]);
       
   230	  // When any location changes, update destination_id as a stringified location object
   231	  React.useEffect(() => {
   232	    if (county && subCounty && ward && gps) {
   233	      setForm((prev) => ({
   234	        ...prev,
   235	        destination_id: JSON.stringify({ county, subCounty, ward, gps }),
   236	      }));
   237	    }
   238	  }, [county, subCounty, ward, gps]);
       
   239	  // Modal scrollable and responsive
   240	  return (
   241	    <Dialog open={open} onOpenChange={onOpenChange}>
   242	      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   243	        <DialogHeader>
   244	          <DialogTitle>Create Attraction</DialogTitle>
   245	          <DialogDescription>Add a new attraction site to operations tracking.</DialogDescription>
   246	        </DialogHeader>
       
   247	        <form
   248	          className="grid gap-3 md:grid-cols-2"
   249	          onSubmit={(event) => {
   250	            event.preventDefault();
   251	            onSubmit();
   252	          }}
   253	        >
   254	          <label className="text-sm md:col-span-2">
   255	            Name
   256	            <input
   257	              required
   258	              value={form.name}
   259	              onChange={(event) => setForm((prev) => ({ ...prev, name: event.target.value }))}
   260	              className="mt-1 w-full rounded-md border border-border bg-background px-3 py-2"
   261	            />
   262	          </label>
       
   263	          <label className="text-sm">
   264	            County
   265	            <select
   266	              required
   267	              value={county}
   268	              onChange={(e) => {
   269	                setCounty(e.target.value);
   270	                setSubCounty("");
   271	                setWard("");

   273	              }}
